// ======================================================================
// 0. CONSTRAINTS / UNIQUENESS
// ======================================================================
// Enforce uniqueness on key node properties so that MERGE can safely
// find-or-create nodes by their business keys (analogous to PRIMARY KEY).

// Equivalent to SQL’s PRIMARY KEY / UNIQUE constraints.
//  We enforce that:
 //   • No two City nodes share the same name.
 //   • No two State nodes share the same name.
 //   • No two Region nodes share the same name.
 //   • No two Customer nodes share the same id.
 //   • No two Product nodes share the same id.
 //   • No two Order nodes share the same id.
 //   • No two Weather nodes share the same (city, date) tuple.
//  This ensures data integrity and allows Neo4j to optimize lookups via indexes.
*/
// ======================================================================

CREATE CONSTRAINT city_name IF NOT EXISTS
  FOR (c:City) REQUIRE c.name IS UNIQUE;

CREATE CONSTRAINT state_name IF NOT EXISTS
  FOR (s:State) REQUIRE s.name IS UNIQUE;

CREATE CONSTRAINT region_name IF NOT EXISTS
  FOR (r:Region) REQUIRE r.name IS UNIQUE;

CREATE CONSTRAINT customer_id IF NOT EXISTS
  FOR (c:Customer) REQUIRE c.id IS UNIQUE;

CREATE CONSTRAINT product_id IF NOT EXISTS
  FOR (p:Product) REQUIRE p.id IS UNIQUE;

CREATE CONSTRAINT order_id IF NOT EXISTS
  FOR (o:Order) REQUIRE o.id IS UNIQUE;

CREATE CONSTRAINT weather_key IF NOT EXISTS
  FOR (w:Weather) REQUIRE (w.city, w.date) IS UNIQUE;



// ======================================================================
// 1. BULK-LOAD “sales_final.csv”
//    - Creates/merges: Region, State, City, Customer, Product, Order nodes.
//    - Creates relationships: IN_REGION, IN_STATE, DELIVERED_TO, PLACED, CONTAINS.
//    - Accumulates sales/profit on Order if same order_id appears multiple times.
// ======================================================================
// Concept: In SQL we would INSERT into multiple tables and link them by foreign keys.
// Here, each `MERGE` on a label and property combines matching or creation,
// and each path (e.g., (cust)-[:PLACED]->(ord)) stands in for a join on customer_id.
// ======================================================================

:auto
CALL {
  WITH "file:///sales_final.csv" AS url
  LOAD CSV WITH HEADERS FROM url AS row

  // Create or match Region by name
  MERGE (reg:Region {name: row.region})

  // Create or match State by name, link to its Region
  MERGE (st:State {name: row.state})
  MERGE (st)-[:IN_REGION]->(reg)

  // Create or match City by name, link to its State
  MERGE (city:City {name: row.city})
  MERGE (city)-[:IN_STATE]->(st)

  // Create or match Customer by id, set properties on creation
  MERGE (cust:Customer {id: row.customer_id})
    ON CREATE SET
      cust.name    = row.customer_name,
      cust.segment = row.segment

  // Create or match Product by id, set properties on creation
  MERGE (prod:Product {id: row.product_id})
    ON CREATE SET
      prod.name         = row.product_name,
      prod.category     = row.category,
      prod.sub_category = row.sub_category

  // Create or match Order by id.
  // On first appearance, set date, sales, profit.
  // On subsequent appearances (same order_id, multiple items), accumulate.
  MERGE (ord:Order {id: row.order_id})
    ON CREATE SET
      ord.date   = date(row.order_date),
      ord.sales  = toFloat(row.sales),
      ord.profit = toFloat(row.profit)
    ON MATCH SET
      ord.sales  = ord.sales  + toFloat(row.sales),
      ord.profit = ord.profit + toFloat(row.profit)

  // Link Customer→Order
  MERGE (cust)-[:PLACED]->(ord)

  // Link Order→City (delivery location)
  MERGE (ord)-[:DELIVERED_TO]->(city)

  // Link Order→Product with quantity and discount stored on the relationship
  MERGE (ord)-[:CONTAINS {qty: toInteger(row.quantity), discount: toFloat(row.discount)}]->(prod)
}
IN TRANSACTIONS OF 5000 ROWS;



// ======================================================================
// 2. BULK-LOAD “description.csv”
//    - Creates/merges: Weather nodes with (city, date) as unique key.
//    - Stores the description property on Weather.
//    - Creates HAS_WEATHER relationship City→Weather.
// ======================================================================
// Concept: Instead of inserting into a Weather table and joining later,
// here we MATCH City nodes by name and MERGE Weather nodes directly.
// The pattern (city)-[:HAS_WEATHER]->(w:Weather) connects them.
// ======================================================================

:auto
CALL {
  WITH "file:///description.csv" AS url
  LOAD CSV WITH HEADERS FROM url AS row

  // Find the existing City node by name
  MATCH (city:City {name: row.city})

  // Create or match a Weather node by (city, date)
  // On creation, set its description property
  MERGE (w:Weather {city: row.city, date: date(row.date)})
    ON CREATE SET w.description = row.description

  // Link City→Weather
  MERGE (city)-[:HAS_WEATHER]->(w)
}
IN TRANSACTIONS OF 10000 ROWS;



// ======================================================================
// 3. BULK-LOAD “temperature1.csv”
//    - Updates Weather.temperature by converting Kelvin→Celsius.
// ======================================================================
// Concept: In SQL: UPDATE Weather SET temperature = temperature - 273.15 WHERE ....
// Here, we MATCH Weather nodes and SET the property.
// ======================================================================

:auto
CALL {
  WITH "file:///temperature1.csv" AS url
  LOAD CSV WITH HEADERS FROM url AS row

  // Match Weather by composite key (city, date)
  MATCH (w:Weather {city: row.city, date: date(row.date)})

  // Convert Kelvin to Celsius and assign
  SET w.temperature = toFloat(row.temperature) - 273.15
}
IN TRANSACTIONS OF 10000 ROWS;



// ======================================================================
// 4. BULK-LOAD “humidity.csv”
//    - Updates Weather.humidity property.
// ======================================================================
// Concept: Same as temperature: MATCH then SET.
// ======================================================================

:auto
CALL {
  WITH "file:///humidity.csv" AS url
  LOAD CSV WITH HEADERS FROM url AS row

  // Match Weather by (city, date)
  MATCH (w:Weather {city: row.city, date: date(row.date)})

  // Assign humidity
  SET w.humidity = toFloat(row.humidity)
}
IN TRANSACTIONS OF 10000 ROWS;



// ======================================================================
// 5. ANALYTIC QUERIES
//    Reformulated from SQL into Cypher, with high-level commentary.
// ======================================================================

//
// Q1: Average Temperature & Humidity per City
//
// We traverse one hop: City→HAS_WEATHER→Weather. Neo4j follows that edge directly,
// rather than performing an SQL JOIN on City.name = Weather.city.
// Grouping happens implicitly by returning c.name + avg(...) together.
//
MATCH (c:City)-[:HAS_WEATHER]->(w:Weather)
RETURN
  c.name                       AS city,
  round(avg(w.temperature), 2) AS avg_temp,
  round(avg(w.humidity),    2) AS avg_humidity
ORDER BY city;



//
// Q2: Average Sales by Region and City
//
// We describe the path: Region<-IN_REGION-State<-IN_STATE-City<-DELIVERED_TO-Order.
// In SQL this would be four JOINs. Here it’s one MATCH with chained relationships.
// Aggregation by region & city is implicit.
//
MATCH (r:Region)<-[:IN_REGION]-(s:State)<-[:IN_STATE]-(c:City)<-[:DELIVERED_TO]-(o:Order)
RETURN
  r.name                 AS region,
  c.name                 AS city,
  round(avg(o.sales), 2) AS avg_sales
ORDER BY region, avg_sales DESC;



//
// Q3: Total Sales & Profit by Region and State
//
// Path: Region<-IN_REGION-State<-IN_STATE-City<-DELIVERED_TO-Order.
// Implicit grouping by returning r.name + s.name + sum(...).
//
MATCH (r:Region)<-[:IN_REGION]-(s:State)<-[:IN_STATE]-(:City)<-[:DELIVERED_TO]-(o:Order)
RETURN
  r.name                  AS region,
  s.name                  AS state,
  round(sum(o.sales),  2) AS total_sales,
  round(sum(o.profit), 2) AS total_profit
ORDER BY total_sales DESC;



//
// Q4: Most Profitable Customers
//
// Path: Customer-[:PLACED]->Order. No JOIN clause needed—pattern matching suffices.
// Then sum profits per customer implicitly.
//
MATCH (cust:Customer)-[:PLACED]->(o:Order)
RETURN
  cust.id                 AS customer_id,
  cust.name               AS customer_name,
  round(sum(o.profit), 2) AS total_profit
ORDER BY total_profit DESC
LIMIT 10;



//
// Q5: Product Performance by Segment, Category, Subcategory
//
// Path: Customer-[:PLACED]->Order-[:CONTAINS]->Product.
// The CONTAINS relationship replaces a join‐table. We aggregate sales/profit per segment/category.
// Grouping is implicit when returning segment + category + sum(...).
//
MATCH (cust:Customer)-[:PLACED]->(o:Order)-[:CONTAINS]->(p:Product)
RETURN
  cust.segment                    AS segment,
  p.category                      AS category,
  p.sub_category                  AS sub_category,
  count(o)                        AS number_of_orders,
  round(sum(o.sales),  2)         AS total_sales,
  round(sum(o.profit), 2)         AS total_profit
ORDER BY segment, total_sales DESC;



//
// Q6: Weather on High‐Sales Days (> $1000)
//
// We first match two hops: City<-DELIVERED_TO-Order and City-[:HAS_WEATHER]->Weather.
// Then in WITH, we group by city and date(o.date), sum sales, and carry along Weather node.
// Filtering by total_sales > 1000 and matching w.date = order_date replaces SQL’s HAVING.
//
MATCH (c:City)<-[:DELIVERED_TO]-(o:Order),
      (c)-[:HAS_WEATHER]->(w:Weather)
WITH
  c,
  date(o.date)   AS d,
  sum(o.sales)   AS total_sales,
  w
WHERE
  total_sales > 1000
  AND w.date = d
RETURN
  c.name                  AS city,
  d                       AS order_date,
  w.description           AS weather_description
ORDER BY total_sales DESC;



//
// Q7: Top‐Selling Products on Sunny Days per Region
//
// Path: Region<-IN_REGION-State<-IN_STATE-City-[:HAS_WEATHER]->Weather
//      AND City<-DELIVERED_TO-Order-[:CONTAINS]->Product.
// We filter Weather.description CONTAINS "sky is clear" and match w.date = o.date.
// Then group sales per (region, product), sort, and pick the first entry per region via collect()[0].
// In SQL, one might have used a window function; here collect() + slicing is the graph‐native pattern.
//
MATCH (r:Region)<-[:IN_REGION]-(:State)<-[:IN_STATE]-(c:City),
      (c)-[:HAS_WEATHER]->(w:Weather),
      (c)<-[:DELIVERED_TO]-(o:Order)-[:CONTAINS]->(p:Product)
WHERE
  w.description CONTAINS "sky is clear"
  AND w.date = o.date
WITH
  r.name           AS region,
  p.name           AS product,
  sum(o.sales)     AS total_sales
ORDER BY
  region, total_sales DESC
WITH
  region,
  collect({product: product, sales: total_sales})[0] AS top_product
RETURN
  region,
  top_product.product AS product_name,
  top_product.sales   AS sales_value;



//
// Q8: Underperforming Customers vs Regional Profit
//
// First MATCH calculates each customer’s avg profit per region.
// Second MATCH recalculates each region’s avg profit across all customers.
// Then filter where cust_avg < region_avg.
// In SQL, two CTEs would be used; here we chain MATCH … WITH blocks.
// Relationships implicitly traverse foreign‐key links.
//
MATCH (cust:Customer)-[:PLACED]->(o:Order)-[:DELIVERED_TO]->(:City)-[:IN_STATE]->(:State)-[:IN_REGION]->(r:Region)
WITH
  r,
  cust,
  avg(o.profit) AS cust_avg
MATCH (:Customer)-[:PLACED]->(ord:Order)-[:DELIVERED_TO]->(:City)-[:IN_STATE]->(:State)-[:IN_REGION]->(r)
WITH
  r,
  cust,
  cust_avg,
  avg(ord.profit) AS region_avg
WHERE
  cust_avg < region_avg
RETURN
  cust.id                   AS customer_id,
  cust.name                 AS customer_name,
  r.name                    AS region,
  round(cust_avg, 2)        AS customer_avg_profit,
  round(region_avg, 2)      AS region_avg_profit
ORDER BY region_avg DESC;



//
// Q9: Monthly Sales & Avg Temperature per City (Year: 2014)
//
// Path: City-[:DELIVERED_TO]-Order and City-[:HAS_WEATHER]-Weather.
// Filter on year via o.date.year = 2014 and match w.date = o.date.
// Grouping by c.name and o.date.month is implicit in RETURN.
// In SQL, we’d use GROUP BY and EXTRACT(MONTH ...).
//
MATCH (c:City)<-[:DELIVERED_TO]-(o:Order),
      (c)-[:HAS_WEATHER]->(w:Weather)
WHERE
  o.date.year = 2014
  AND w.date = o.date
WITH
  c.name                   AS city,
  o.date.month             AS month,
  sum(o.sales)             AS total_sales,
  avg(w.temperature)       AS avg_temp
RETURN
  city,
  month,
  round(total_sales, 2) AS total_sales,
  round(avg_temp,    1) AS avg_temp
ORDER BY city, month;



//
// Q10: Top 5 Best‐Selling Products per Weather Condition
//
// Path: Order-[:CONTAINS]-Product and Order-[:DELIVERED_TO]-City-[:HAS_WEATHER]-Weather.
// Filter on o.date = w.date, then aggregate sales by (weather, product).
// Sort by weather + total_sales DESC, then collect top 5 per weather.
// In SQL, this would be a PARTITION BY + RANK() window function.
// Here, collect()[0..5] is the graph equivalent.
//
MATCH (o:Order)-[:CONTAINS]->(p:Product),
      (o)-[:DELIVERED_TO]->(c:City)-[:HAS_WEATHER]->(w:Weather)
WHERE
  o.date = w.date
WITH
  w.description   AS weather,
  p.name          AS product,
  sum(o.sales)    AS total_sales
ORDER BY
  weather, total_sales DESC
WITH
  weather,
  collect({product: product, sales: total_sales})[0..5] AS top5
UNWIND top5 AS row
RETURN
  weather,
  row.product         AS product_name,
  round(row.sales, 2) AS total_sales
ORDER BY
  weather, total_sales DESC;
